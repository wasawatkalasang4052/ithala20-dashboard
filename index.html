#define TG_TOKEN "123456789:AAxxxxxxxxxxxxxxxxxxxxxx" #define TG_CHAT "987654321"
2. sendTelegram()
void sendTelegram(String msg) { if (WiFi.status() != WL_CONNECTED) return; HTTPClient http; String url = "https://api.telegram.org/bot" + String(TG_TOKEN) + "/sendMessage?chat_id=" + String(TG_CHAT) + "&text=" + urlencode(msg); http.begin(url); http.GET(); http.end(); }
3. ใน loop() + threshold ที่ตั้งไว้
if (rain_raw < 2000 && !wasRaining) { sendTelegram("🌧 ITHALA20: ฝนเริ่มตก (rain=" + String(rain_raw) + ")"); wasRaining = true; } if (temp > THRESH_TEMP) { sendTelegram("🌡 ร้อนมาก! " + String(temp) + "°C"); }
